% Cover: the brand ground with the title in the open space, the subtitle beneath it and the author at the foot. It reads
% the title, subtitle, author and date of the book from the project metadata. Spacing uses \par and \vspace because a
% line break after an empty field, such as an unset subtitle, fails with "There's no line here to end".
% A finished cover, with its own title and author, is a PDF at assets/cover-full.pdf: it is shown alone, full page, and nothing
% is printed over it. Without it, the cover is built from the brand ground, an optional picture at assets/cover.pdf and the
% title, subtitle and author of the book.
\makeatletter
\xdef\booktitle{\@title}
\IfFileExists{assets/cover-full.pdf}{%
  \begin{titlepage}
  \thispagestyle{bkcover}
  \AddToShipoutPictureBG*{%
    \includegraphics[width=\paperwidth, height=\paperheight]{assets/cover-full.pdf}}
  \null
  \end{titlepage}
}{%
\providecommand{\@subtitle}{}
\begin{titlepage}
\thispagestyle{bkcover}
\AddToShipoutPictureBG*{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[knInk] (current page.south west) rectangle (current page.north east);
    % An optional full-bleed cover picture: put a PDF at assets/cover.pdf in the project.
    \IfFileExists{assets/cover.pdf}{%
      \node[anchor=south west, inner sep=0pt] at (current page.south west) {%
        \includegraphics[width=\paperwidth, height=\paperheight]{assets/cover.pdf}};
    }{}
  \end{tikzpicture}%
}
\begin{tikzpicture}[remember picture, overlay]
  % A short rule in the book blue, then the title in the open sky.
  \node[anchor=north west, xshift=2.6cm, yshift=-2.4cm] at (current page.north west) {%
    \begin{minipage}{15.8cm}
      \raggedright
      {\color{knSkyBright}\rule{2.2cm}{2.5pt}}\par
      \vspace{22pt}
      {\bkCoverFont\fontsize{40}{45}\selectfont\color{white}\@title}\par
      \vspace{12pt}
      % \rmfamily, not \bkHeadingFont: the heading family is pinned to its
      % weight through a RawFeature axis, so anything set in it arrives at 600.
      {\rmfamily\fontsize{14}{18}\selectfont\color{knSkyBright}\@subtitle}\par
    \end{minipage}};
  % Author on the ground below the street; the site on the right.
  \node[anchor=south west, xshift=2.6cm, yshift=1.0cm] at (current page.south west) {%
    \begin{minipage}{12cm}
      \raggedright
      {\bkHeadingFont\fontsize{13}{16}\selectfont\color{white}\@author}\par
    \end{minipage}};
  \node[anchor=south east, xshift=-2.6cm, yshift=1.0cm] at (current page.south east) {%
    {\rmfamily\fontsize{8.5}{10}\selectfont\color{white!60}\bookweb}};
\end{tikzpicture}
\end{titlepage}
}
\makeatother
