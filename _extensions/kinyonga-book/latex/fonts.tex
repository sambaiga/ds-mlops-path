% Fonts of the print edition: Libre Franklin for text and Jost for headings, the pair of the web edition, loaded from
% the files that Quarto copies next to the generated .tex (format-resources), so no path is needed and a render from a
% sub-folder works. Code uses JetBrains Mono.
\usepackage{fontspec}

\setmainfont{LibreFranklin-Regular.ttf}[
  ItalicFont     = LibreFranklin-Italic.ttf,
  BoldFont       = LibreFranklin-SemiBold.ttf,
  BoldItalicFont = LibreFranklin-SemiBoldItalic.ttf,
]

\setsansfont{LibreFranklin-Regular.ttf}[
  ItalicFont     = LibreFranklin-Italic.ttf,
  BoldFont       = LibreFranklin-SemiBold.ttf,
  BoldItalicFont = LibreFranklin-SemiBoldItalic.ttf,
]

\newfontfamily\bkHeadingFont{Jost-SemiBold.ttf}[BoldFont = Jost-Bold.ttf]
\newfontfamily\bkCoverFont{Jost-Bold.ttf}[]

% Code in JetBrains Mono, slightly scaled to sit with the text face.
\setmonofont{JetBrainsMono-Regular.ttf}[BoldFont = JetBrainsMono-Bold.ttf, Scale = 0.86]
